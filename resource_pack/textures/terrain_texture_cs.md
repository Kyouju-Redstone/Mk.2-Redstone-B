# terrain_texture.json cheat sheet

## Hopper
### Texture assignment
```
"hopper_top": {
	"textures": [
		"textures/blocks/hopper/top_downward",		// down		enabled=false
		"textures/blocks/hopper/top",				// unused
		"textures/blocks/hopper/top_north",			// north	enabled=false
		"textures/blocks/hopper/top_south",			// south	enabled=false
		"textures/blocks/hopper/top_west",			// west		enabled=false
		"textures/blocks/hopper/top_east",			// east		enabled=false
		"textures/blocks/hopper/top",				// unused
		"textures/blocks/hopper/top",				// unused
		"textures/blocks/hopper/top_downward_on",	// down 	enabled=true
		"textures/blocks/hopper/top",				// unused
		"textures/blocks/hopper/top_north_on",		// north	enabled=true
		"textures/blocks/hopper/top_south_on",		// south	enabled=true
		"textures/blocks/hopper/top_west_on",		// west		enabled=true
		"textures/blocks/hopper/top_east_on"		// east		enabled=true
	]
}
```
